def maxTint(c,d):
    d = map(set,d)
    for ci in range(len(c)):
        new_d = []
        new_g = []
        for di in d:
            if ci in di:
                new_g.append(di)
            else:
                new_d.append(di)
        d = new_d
        if len(new_g)!=0:
            d.append(set())
            for gi in new_g:
                d[-1]=d[-1]|gi
    ans = 0
    for di in d:
        ansi = 0
        for i in di:
            ansi=max(ansi,c.count(c[i-1]))
        ans+=ansi
    return ans

时间复杂度：n*m
